## 데이터 집계와 분석
- 여러 행을 하나의 숫자로 요약하는 집계함수를 쓰고, GROUP BY로 월별·강좌별로 묶어 비교합니다. WHERE와 HAVING이 각각 무엇을 거르는지 구분합니다.

``` sql
    SELECT  gender, 
            COUNT(*) AS cnt, 
            ROUND(AVG(birth_year), 2) AS avg_year
    FROM members
    WHERE member_id <= 7
    AND birth_year >= 1980
    GROUP BY gender
    HAVING COUNT(*) >= 3
    ORDER BY gender;
```

SELECT course_name, level , start_time 
FROM courses

SELECT name, birth_year,   
        CASE WHEN birth_year <= 1960 THEN '할인'
             WHEN birth_year >= 2000 THEN '청소년'
             ELSE '일반'
        END AS 요금구분
FROM members
ORDER BY birth_year , name
LIMIT 10;   #CASE 문은 데이터가 많아지면 속도가 떨어짐

SELECT

    MIN(fee)                 AS 최저수강료 ,
    MAX(fee)                 AS 최고수강료 ,

    ROUND(AVG(fee),2)        AS 평균수강료
FROM courses;

SELECT
    COUNT(*)            AS 신청건수 ,
    SUM(pay_amount)     AS 매출     ,
    MAX(pay_amount)     AS 최고매출
FROM enrollments
WHERE enroll_month = '2026-11'; 

SELECT 
    enroll_month       AS 신청일   ,
    COUNT(*)            AS 신청건수 ,
    SUM(pay_amount)     AS 매출
FROM enrollments
GROUP BY enroll_month
ORDER BY enroll_month;

SELECT
    enroll_month, status ,COUNT(*)  AS 건수
FROM enrollments
GROUP BY enroll_month, status
ORDER BY enroll_month, status;


-- HAVING

SELECT course_id, COUNT(*)     AS 누적신청
FROM enrollments
WHERE status <> '취소'
GROUP BY course_id
HAVING COUNT(*) >= 18
ORDER BY 누적신청 DESC, course_id;

SELECT  region          AS 거주지,
        COUNT(*)        AS 회원수

HAVING COUNT(*) >= 10
ORDER BY 회원수 DESC;

SELECT 
    e.member_id,
    m.name,
    c.course_name,
    e.enroll_month,
    e.status
FROM enrollments e
LEFT JOIN members m ON e.member_id = m.member_id
LEFT JOIN courses c ON e.course_id = c.course_id
WHERE e.course_id = 1 AND e.enroll_month = '2026-11'
ORDER BY m.name;

SELECT paid.member_id   AS 회원번호 , 
       m.name           AS 회원명   ,
       paid.total_paid  AS 매출액   
FROM (
    SELECT member_id, SUM(pay_amount) AS total_paid
    FROM enrollments
    GROUP BY member_id
) AS paid
LEFT JOIN members m ON paid.member_id = m.member_id 
WHERE paid.total_paid >= 400000
ORDER BY paid.total_paid DESC, paid.member_id
